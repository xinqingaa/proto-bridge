class AccountHoldingProtoPage extends StatefulWidget {
  const AccountHoldingProtoPage({super.key});

  @override
  State<AccountHoldingProtoPage> createState() =>
      _AccountHoldingProtoPageState();
}

class _AccountHoldingProtoPageState extends State<AccountHoldingProtoPage> {
  static const _repository = AccountProtoRepository();
  HoldingProtoFilter _filter = HoldingProtoFilter.all;

  List<HoldingProtoPosition> get _visibleHoldings {
    final holdings = _repository.holdings;
    if (_filter == HoldingProtoFilter.all) {
      return holdings;
    }
    return holdings.where((item) => item.filter == _filter).toList();
  }

  @override
  Widget build(BuildContext context) {
    final summary = _repository.holdingSummary;
    return Scaffold(
      appBar: CommonAppBar(
        title: context.t('asset.holdings.title'),
        eyebrow: context.t('asset.holdings.subtitle'),
        leading: ProtoIconButton(
          icon: Icons.chevron_left,
          tooltip: context.t('asset.action.back'),
          onPressed: () => Navigator.of(context).maybePop(),
        ),
        action: ProtoIconButton(
          icon: Icons.refresh,
          tooltip: context.t('asset.action.refresh'),
          onPressed: () => setState(() {}),
        ),
      ),
      body: ListView(
        padding: const EdgeInsets.fromLTRB(
          AppSpacing.md,
          AppSpacing.xs,
          AppSpacing.md,
          AppSpacing.lg,
        ),
        children: [
          _HoldingSummaryCard(summary: summary),
          const SizedBox(height: AppSpacing.lg),
          Wrap(
            spacing: AppSpacing.xs,
            runSpacing: AppSpacing.xs,
            children: [
              _holdingChip(HoldingProtoFilter.all, context.t('filter.all')),
              _holdingChip(
                HoldingProtoFilter.semiconductor,
                context.t('sector.semiconductor'),
              ),
              _holdingChip(
                HoldingProtoFilter.consumer,
                context.t('sector.consumer'),
              ),
              _holdingChip(HoldingProtoFilter.ev, context.t('sector.ev')),
            ],
          ),
          const SizedBox(height: AppSpacing.lg),
          _HoldingListPanel(holdings: _visibleHoldings),
          const SizedBox(height: AppSpacing.lg),
          Row(
            children: [
              Expanded(
                child: CommonButton(
                  label: context.t('asset.action.analysis'),
                  onPressed: () =>
                      Navigator.of(context).pushNamed(Routes.pnlAnalysis),
                ),
              ),
              const SizedBox(width: AppSpacing.sm),
              Expanded(
                child: CommonButton(
                  label: context.t('asset.action.rebalance'),
                  primary: true,
                  onPressed: () {},
                ),
              ),
            ],
          ),
        ],
      ),
    );
  }

  Widget _holdingChip(HoldingProtoFilter filter, String label) {
    return ProtoFilterChip(
      label: label,
      selected: _filter == filter,
      onTap: () => setState(() => _filter = filter),
    );
  }
}

class AccountPnlProtoPage extends StatefulWidget {
  const AccountPnlProtoPage({
    this.initialTab = PnlProtoTab.overview,
    super.key,
  });

  final PnlProtoTab initialTab;

  @override
  State<AccountPnlProtoPage> createState() => _AccountPnlProtoPageState();
}

class _AccountPnlProtoPageState extends State<AccountPnlProtoPage> {
  static const _repository = AccountProtoRepository();
  late PnlProtoTab _tab;
  RiskProtoFilter _riskFilter = RiskProtoFilter.all;

  List<PnlProtoRecord> get _filteredRecords {
    final records = _repository.records;
    if (_riskFilter == RiskProtoFilter.all) {
      return records;
    }
    return records.where((record) => record.risk == _riskFilter).toList();
  }

  List<PnlProtoRecord> get _realizedRecords {
    return _repository.records
        .where((record) => record.tone != ProtoTone.negative)
        .toList();
  }

  @override
  void initState() {
    super.initState();
    _tab = widget.initialTab;
    if (_tab == PnlProtoTab.risk) {
      _riskFilter = RiskProtoFilter.high;
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: CommonAppBar(
        title: context.t('asset.pnl.title'),
        eyebrow: context.t('app.brand'),
        leading: ProtoIconButton(
          icon: Icons.chevron_left,
          tooltip: context.t('asset.action.back'),
          onPressed: () => Navigator.of(context).maybePop(),
        ),
        action: ProtoIconButton(
          icon: Icons.keyboard_command_key,
          tooltip: context.t('asset.action.filter'),
          onPressed: _openRiskFilterSheet,
        ),
      ),
      body: ListView(
        padding: const EdgeInsets.fromLTRB(
          AppSpacing.md,
          AppSpacing.xs,
          AppSpacing.md,
          AppSpacing.lg,
        ),
        children: [
          ProtoTabBar(activeTab: _tab, onChanged: _setTab),
          const SizedBox(height: AppSpacing.sm),
          switch (_tab) {
            PnlProtoTab.overview => _OverviewTab(
                metrics: _repository.pnlMetrics,
                trendPoints: _repository.trendPoints,
                records: _filteredRecords,
                riskFilter: _riskFilter,
                onFilter: _openRiskFilterSheet,
                onRecordTap: _openRecordSheet,
              ),
            PnlProtoTab.realized => _RealizedTab(
                breakdown: _repository.realizedBreakdown,
                records: _realizedRecords,
                onFilter: _openRiskFilterSheet,
                onRecordTap: _openRecordSheet,
              ),
            PnlProtoTab.risk => _RiskTab(
                exposureItems: _repository.exposureItems,
                records: _filteredRecords,
                riskFilter: _riskFilter,
                onFilter: _openRiskFilterSheet,
                onRecordTap: _openRecordSheet,
              ),
          },
        ],
      ),
    );
  }

  void _setTab(PnlProtoTab tab) {
    setState(() {
      _tab = tab;
      if (tab == PnlProtoTab.risk) {
        _riskFilter = RiskProtoFilter.high;
      } else if (_riskFilter == RiskProtoFilter.high &&
          tab == PnlProtoTab.overview) {
        _riskFilter = RiskProtoFilter.all;
      }
    });
  }

  void _openRiskFilterSheet() {
    showModalBottomSheet<void>(
      context: context,
      showDragHandle: true,
      backgroundColor: context.pbColors.surface,
      builder: (sheetContext) {
        return Padding(
          padding: const EdgeInsets.fromLTRB(
            AppSpacing.md,
            0,
            AppSpacing.md,
            AppSpacing.lg,
          ),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Text(
                context.t('pnl.riskFilter'),
                style: context.pbTextStyles.sectionTitle,
              ),
              const SizedBox(height: AppSpacing.sm),
              for (final filter in RiskProtoFilter.values) ...[
                SizedBox(
                  width: double.infinity,
                  child: ProtoFilterChip(
                    label: riskProtoLabel(context, filter),
                    selected: _riskFilter == filter,
                    onTap: () {
                      setState(() => _riskFilter = filter);
                      Navigator.of(sheetContext).pop();
                    },
                  ),
                ),
                const SizedBox(height: AppSpacing.xs),
              ],
            ],
          ),
        );
      },
    );
  }

  void _openRecordSheet(PnlProtoRecord record) {
    showModalBottomSheet<void>(
      context: context,
      showDragHandle: true,
      backgroundColor: context.pbColors.surface,
      builder: (sheetContext) {
        final colors = context.pbColors;
        return Padding(
          padding: const EdgeInsets.fromLTRB(
            AppSpacing.md,
            0,
            AppSpacing.md,
            AppSpacing.lg,
          ),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Text(record.symbol, style: context.pbTextStyles.title),
              const SizedBox(height: AppSpacing.xs),
              Text(
                '${context.t(record.actionKey)} · ${record.time}',
                style: TextStyle(color: colors.muted),
              ),
              const SizedBox(height: AppSpacing.md),
              Row(
                children: [
                  Expanded(
                    child: _DetailTile(
                      label: context.t('record.amount'),
                      value: record.amount,
                      color: protoToneColor(context, record.tone),
                    ),
                  ),
                  const SizedBox(width: AppSpacing.xs),
                  Expanded(
                    child: _DetailTile(
                      label: context.t('record.risk'),
                      value: riskProtoLabel(context, record.risk),
                    ),
                  ),
                ],
              ),
              const SizedBox(height: AppSpacing.md),
              CommonButton(
                label: context.t('asset.action.close'),
                primary: true,
                onPressed: () => Navigator.of(sheetContext).pop(),
              ),
            ],
          ),
        );
      },
    );
  }
}
