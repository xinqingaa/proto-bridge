enum ProtoTone { positive, negative, neutral }

enum HoldingProtoFilter { all, semiconductor, consumer, ev }

enum PnlProtoTab { overview, realized, risk }

enum RiskProtoFilter { all, low, medium, high }

class HoldingProtoSummary {
  const HoldingProtoSummary({
    required this.marketValue,
    required this.dayPnl,
    required this.dayPnlRate,
    required this.riskKey,
  });

  final String marketValue;
  final String dayPnl;
  final String dayPnlRate;
  final String riskKey;
}

class HoldingProtoPosition {
  const HoldingProtoPosition({
    required this.symbol,
    required this.name,
    required this.amount,
    required this.shares,
    required this.pnl,
    required this.pnlRate,
    required this.filter,
    required this.risk,
    required this.tone,
  });

  final String symbol;
  final String name;
  final String amount;
  final int shares;
  final String pnl;
  final String pnlRate;
  final HoldingProtoFilter filter;
  final RiskProtoFilter risk;
  final ProtoTone tone;
}

class PnlProtoMetric {
  const PnlProtoMetric({
    required this.labelKey,
    required this.value,
    required this.delta,
    required this.tone,
  });

  final String labelKey;
  final String value;
  final String delta;
  final ProtoTone tone;
}

class PnlProtoRecord {
  const PnlProtoRecord({
    required this.id,
    required this.symbol,
    required this.actionKey,
    required this.time,
    required this.amount,
    required this.risk,
    required this.tone,
  });

  final String id;
  final String symbol;
  final String actionKey;
  final String time;
  final String amount;
  final RiskProtoFilter risk;
  final ProtoTone tone;
}

class RealizedProtoBreakdown {
  const RealizedProtoBreakdown({
    required this.label,
    required this.value,
    required this.tone,
  });

  final String label;
  final String value;
  final ProtoTone tone;
}

class ExposureProtoItem {
  const ExposureProtoItem({
    required this.labelKey,
    required this.value,
  });

  final String labelKey;
  final String value;
}
