import 'package:flutter/material.dart';
import 'package:flutter_bloc/flutter_bloc.dart';

import 'preferences/app_preferences_cubit.dart';
import 'routes/app_pages_proto.dart';
import 'routes/app_routes.dart';
import 'theme/app_theme.dart';

class ProtoBridgeApp extends StatelessWidget {
  const ProtoBridgeApp({super.key});

  @override
  Widget build(BuildContext context) {
    return BlocProvider(
      create: (_) => AppPreferencesCubit(),
      child: BlocBuilder<AppPreferencesCubit, AppPreferencesState>(
        builder: (context, state) {
          return MaterialApp(
            debugShowCheckedModeBanner: false,
            title: 'ProtoBridge',
            theme: buildAppTheme(),
            darkTheme: buildAppTheme(brightness: Brightness.dark),
            themeMode: state.themeMode,
            locale: state.isEnglish
                ? const Locale('en', 'US')
                : const Locale('zh', 'CN'),
            initialRoute: Routes.home,
            onGenerateRoute: AppPagesProto.onGenerateRoute,
          );
        },
      ),
    );
  }
}
