Color protoToneColor(BuildContext context, ProtoTone tone) {
  final colors = context.pbColors;
  return switch (tone) {
    ProtoTone.positive => colors.positive,
    ProtoTone.negative => colors.negative,
    ProtoTone.neutral => colors.text,
  };
}

String riskProtoLabel(BuildContext context, RiskProtoFilter risk) {
  return switch (risk) {
    RiskProtoFilter.all => context.t('risk.all'),
    RiskProtoFilter.low => context.t('risk.low'),
    RiskProtoFilter.medium => context.t('risk.medium'),
    RiskProtoFilter.high => context.t('risk.high'),
  };
}

class ProtoTrendChart extends StatelessWidget {
  const ProtoTrendChart({required this.points, super.key});

  final List<double> points;

  @override
  Widget build(BuildContext context) {
    final colors = context.pbColors;
    final maxPoint = points.reduce((a, b) => a > b ? a : b);
    return DecoratedBox(
      decoration: BoxDecoration(
        color: colors.surfaceSoft,
        borderRadius: BorderRadius.circular(AppRadii.panel),
      ),
      child: SizedBox(
        height: 132,
        child: Padding(
          padding: const EdgeInsets.all(AppSpacing.sm),
          child: Row(
            crossAxisAlignment: CrossAxisAlignment.end,
            children: [
              for (final point in points) ...[
                Expanded(
                  child: Align(
                    alignment: Alignment.bottomCenter,
                    child: FractionallySizedBox(
                      heightFactor: (point / maxPoint).clamp(0.16, 1),
                      widthFactor: 0.78,
                      child: DecoratedBox(
                        decoration: BoxDecoration(
                          borderRadius: const BorderRadius.vertical(
                            top: Radius.circular(6),
                            bottom: Radius.circular(3),
                          ),
                          gradient: LinearGradient(
                            begin: Alignment.topCenter,
                            end: Alignment.bottomCenter,
                            colors: [colors.accent, colors.primary],
                          ),
                        ),
                      ),
                    ),
                  ),
                ),
                if (point != points.last) const SizedBox(width: AppSpacing.xs),
              ],
            ],
          ),
        ),
      ),
    );
  }
}

class ProtoRecordPanel extends StatelessWidget {
  const ProtoRecordPanel({
    required this.title,
    required this.subtitle,
    required this.records,
    required this.onFilter,
    required this.onRecordTap,
    this.loading = false,
    super.key,
  });

  final String title;
  final String subtitle;
  final List<PnlProtoRecord> records;
  final bool loading;
  final VoidCallback onFilter;
  final ValueChanged<PnlProtoRecord> onRecordTap;

  @override
  Widget build(BuildContext context) {
    return SectionPanel(
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          ProtoSectionHeading(title: title, subtitle: subtitle),
          const SizedBox(height: AppSpacing.md),
          SizedBox(
            width: 96,
            child: CommonButton(
              label: context.t('asset.action.filter'),
              onPressed: onFilter,
            ),
          ),
          if (loading)
            CommonLoading(message: context.t('pnl.loading'))
          else if (records.isEmpty)
            CommonEmpty(message: context.t('pnl.empty'))
          else
            for (final record in records)
              _ProtoRecordRow(record: record, onTap: () => onRecordTap(record)),
        ],
      ),
    );
  }
}

class _ProtoRecordRow extends StatelessWidget {
  const _ProtoRecordRow({required this.record, required this.onTap});

  final PnlProtoRecord record;
  final VoidCallback onTap;

  @override
  Widget build(BuildContext context) {
    final colors = context.pbColors;
    return InkWell(
      onTap: onTap,
      child: Padding(
        padding: const EdgeInsets.symmetric(vertical: 10),
        child: Row(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Expanded(
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    '${record.symbol}${context.t(record.actionKey)} · '
                    '${record.time}',
                    style: context.pbTextStyles.bodyStrong,
                  ),
                  const SizedBox(height: AppSpacing.xxs),
                  Text(
                    '${record.amount}${riskProtoLabel(context, record.risk)}',
                    style: context.pbTextStyles.bodyStrong.copyWith(
                      color: protoToneColor(context, record.tone),
                    ),
                  ),
                ],
              ),
            ),
            Icon(Icons.chevron_right, color: colors.muted, size: 18),
          ],
        ),
      ),
    );
  }
}
